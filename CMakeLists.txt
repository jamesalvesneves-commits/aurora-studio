cmake_minimum_required(VERSION 3.24)
project(AuroraStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets)
qt_add_executable(AURORA WIN32 src/cockpit/main.cpp)
target_link_libraries(AURORA PRIVATE Qt6::Widgets)
if(MSVC)
  target_compile_options(AURORA PRIVATE /utf-8 /W4)
endif()
enable_testing()
add_test(NAME cockpit_smoke COMMAND AURORA --smoke-test)
set_tests_properties(cockpit_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
